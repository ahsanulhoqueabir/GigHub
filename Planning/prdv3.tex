\section{Product Overview}

GigHub is a campus-exclusive freelance marketplace for Jagannath University (JnU) students. Every student operates from a single account simultaneously as both buyer and seller --- no role selection required.

\subsection{Core Value Proposition}
\begin{itemize}
    \item Trusted campus-only network with verified student identity
    \item Gig marketplace + job board + peer tutoring in a single platform
    \item Escrow-backed transaction safety for paid services
    \item Real-time communication and review-based trust ratings
    \item Free peer tutoring/tuition exchange with no platform fee
\end{itemize}

\section{Functional Requirements}

\textbf{Notation:} P0 = Must Have, P1 = Should Have

\begin{description}
    \item[\textbf{FR-A}] \textbf{Authentication \& Identity:} Email/password + Google OAuth, secure session tokens, server-side verification, and automatic profile creation.
    \item[\textbf{FR-B}] \textbf{Profile Management:} View/edit profile, avatar upload, skills, public profile, verification badge.
    \item[\textbf{FR-C}] \textbf{Gig Marketplace:} Create/edit gigs with package tiers, images, listings, and basic search/filters.
    \item[\textbf{FR-D}] \textbf{Job Board \& Proposals:} Post jobs, submit/accept proposals; accepted proposals create orders.
    \item[\textbf{FR-E}] \textbf{Orders \& Delivery:} Order lifecycle (pending → in-progress → delivered → completed/disputed), deliveries, revisions, approvals.
    \item[\textbf{FR-F}] \textbf{Payments \& Escrow:} Escrow on payment, release on approval, platform fee (5\% up to BDT 500), payment gateway integration, withdrawals, transaction history.
    \item[\textbf{FR-G}] \textbf{Messaging \& Notifications:} Real-time 1:1 chat, order-linked channels, attachments, in-app notifications and push.
    \item[\textbf{FR-H}] \textbf{Reviews \& Trust:} Mutual reviews and ratings, text feedback, aggregated scores on profiles and listings.
    \item[\textbf{FR-I}] \textbf{Dashboard \& Admin:} Unified user dashboard, earnings and transactions, admin moderation and dispute queue.
    \item[\textbf{FR-J}] \textbf{Tuition Listings:} Free-form tuition listings (title + description), request/accept flow opens chat, no payments involved.
\end{description}

\subsection{Acceptance Criteria}
\begin{itemize}
    \item Core P0 flows (auth, order lifecycle, payments, chat, reviews) work end-to-end and are testable
    \item Financial actions are idempotent and auditable
    \item Public data exposure is restricted to allowed fields
\end{itemize}

\section{Non-Functional Requirements}

\begin{itemize}
    \item \textbf{Performance:} p95 read latency <200ms, chat latency <1s, search <500ms
    \item \textbf{Scalability:} Design for horizontal scaling to support 5,000+ concurrent users
    \item \textbf{Availability:} Target 99.5\% uptime with graceful degradation of non-core features
    \item \textbf{Security:} OWASP controls, token-based auth, input validation, TLS for all traffic
    \item \textbf{Usability:} Mobile-first, bilingual-friendly, aim for WCAG 2.1 AA
    \item \textbf{Maintainability:} Modular services, clear API versioning, replaceable components
    \item \textbf{Observability:} Structured logs, key metrics, and alerting for errors/latency spikes
\end{itemize}

\section{External Integration Requirements}

\subsection{Identity Provider}
\begin{itemize}
    \item Support email/password and Google OAuth through a managed identity provider
    \item Server-side token verification on every protected API call
\end{itemize}

\subsection{Content and Data Management}
\begin{itemize}
    \item Admin and content operations managed through a CMS-backed admin panel
    \item Permission model aligned with platform role definitions
\end{itemize}

\subsection{File and Media Storage}
\begin{itemize}
    \item All user-uploaded content stored in a dedicated cloud object store
    \item File size limit: 25 MB per upload
    \item Signed or access-controlled URLs for sensitive delivery files
\end{itemize}

\subsection{Payment Gateways}
\begin{itemize}
    \item SSLCommerz as primary payment gateway with secure callback handling
    \item All payment callbacks must be validated before triggering financial state changes
\end{itemize}

\subsection{Real-Time Communication}
\begin{itemize}
    \item Real-time bidirectional messaging for chat and live event notifications
    \item Mobile push notifications for key platform events
\end{itemize}

\section{Constraints, Assumptions, and Risks}

\subsection{Constraints}
\begin{itemize}
    \item Platform is campus-exclusive: only JnU students can register
    \item Payments are subject to Bangladesh payment gateway regulations
    \item Platform fee policy: 5\% charge for paid transactions up to BDT 500 (no per-gig override)
\end{itemize}

\subsection{Assumptions}
\begin{itemize}
    \item Students primarily access the platform via Android smartphones
    \item An available admin moderation team from project launch
    \item Payment gateway uptime remains at acceptable commercial levels
    \item JnU student enrollment base provides sufficient initial addressable market
\end{itemize}

\subsection{Risks and Mitigation}

\begin{table}[H]
\centering
\small
\begin{tabular}{|l|l|p{5cm}|}
\hline
\textbf{Risk} & \textbf{Impact} & \textbf{Mitigation} \\
\hline
Payment callback inconsistency & High & Idempotent handlers, reconciliation job, audit alerts \\
Chat or job spam and abuse & Medium & Report flow, moderation queue, rate limits \\
Dispute volume overload & Medium & Explicit dispute policy, SLA tiers, admin queue tooling \\
Student identity misuse & High & Verification flag, strict auth checks, moderation enforcement \\
Low early adoption & Medium & Campus ambassador program, social media push, zero-fee tutoring \\
Payment gateway downtime & Medium & Graceful degradation message, retry queue \\
\hline
\end{tabular}
\end{table}

\subsection{Break-Even Analysis}

\textbf{Monthly Operational Break-Even:}

\[
\text{Transactions needed/month} = \frac{\text{Monthly operating cost}}{\text{Revenue per transaction}} = \frac{70,000}{500 \times 5\%} = \frac{70,000}{25} = 2,800 \text{ transactions}
\]

\[
\text{Active users needed} = \frac{2,800 \text{ tx/month}}{2 \text{ tx/user/month}} = 1,400 \text{ users}
\]

\begin{itemize}
    \item \textbf{Monthly operational break-even:} Approximately Month 13--14 after launch
    \item \textbf{Full investment recovery (cumulative break-even):} Approximately Month 30--32 (roughly mid Year 3)
\end{itemize}

\textbf{5-Year ROI:}

\[
\text{Total Revenue (Y1--Y5)} = \text{BDT } 2,58,44,650
\]

\[
\text{Total Cost (Y0--Y5)} = \text{BDT } 83,75,000
\]

\[
\text{Net 5-Year Profit} = \text{BDT } 1,74,69,650 \approx \text{USD } 1,58,815
\]

\[
\text{ROI (5-year)} = \frac{1,74,69,650}{83,75,000} \times 100 \approx \textbf{209\%}
\]

\subsection{Key Economic Notes}

\begin{itemize}
    \item \textbf{Tutoring is a strategic loss-leader:} Zero-fee tutoring generates no direct revenue but reduces onboarding friction
    \item \textbf{Revenue is conservative:} Projections use only the 5\% service fee
    \item \textbf{Adoption risk is the primary variable:} If Year 1 adoption stays below 700 users, monthly break-even shifts
    \item \textbf{Cost model includes no full-time salaries in operations:} Post-launch maintenance assumes lean part-time team
\end{itemize}
